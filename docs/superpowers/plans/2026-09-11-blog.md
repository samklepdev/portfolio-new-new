# Blog Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a blog at `/blog` and `/blog/[slug]`, adapted from the Radiant Tailwind template onto this repo's existing MDX + Postgres pipeline.

**Architecture:** Long-form post bodies are MDX files in `src/content/posts/`; Postgres holds only the queryable index (title, excerpt, date, status, tags), seeded from those files and joined by `slug`. This mirrors how projects already work. All UI is CSS Modules server components.

**Tech Stack:** Next.js 15 App Router, React 19, TypeScript, Drizzle ORM + postgres-js, `next-mdx-remote/rsc`, `gray-matter`, vitest + @testing-library/react.

**Spec:** `docs/superpowers/specs/2026-09-11-blog-design.md`

## Global Constraints

- **No new dependencies.** Radiant's `dayjs`, `clsx`, `@headlessui`, `@heroicons`, `PortableText` and `@/sanity/*` must not enter this repo. `package.json` is unchanged by this plan.
- **CSS Modules only, never Tailwind.** No `@tailwind` directives, no `tailwind.config.ts`, no postcss/autoprefixer.
- **Never run `npm run build` while `next dev` is running.** Use `npm run build:verify` (writes to `.next-verify`). A plain build deletes the dev server's `layout.css` and strips styling site-wide.
- **Builds and seeds need the DB container up:** `npm run db:up` first. `generateStaticParams` queries Postgres.
- **Never hand-edit files in `drizzle/`.** Schema changes are `src/db/schema.ts` → `npm run db:generate` → `npm run db:migrate`. Commit the generated SQL.
- **The seed must stay idempotent.** Use `onConflictDoUpdate`, never `onConflictDoNothing`, wherever `.returning()` feeds later linking — a do-nothing conflict returns an empty array and the linking below it silently no-ops on re-run.
- **Postgres sorts NULLs first on `DESC`.** Every ordering on a nullable column needs explicit `nulls last` plus a stable tiebreak.
- Colour tokens: `#0B0E14` background, `#12161F` surface, `#00F0FF` turquoise (dominant), `#FF2E97` pink (CTA), `#39FF88` green (status indicators only — not used in this feature).
- Every CSS module ends with `@media (prefers-reduced-motion: reduce)` zeroing its transitions/transforms, and wraps hover-only treatments in `@media (hover: hover) and (pointer: fine)`.
- Run `npm test` before every commit. Run `npm run lint` before the final commit of each task.

```ts
import matter from "gray-matter";

export type Collection = "projects" | "posts";

const CONTENT_ROOT = path.join(process.cwd(), "src", "content");
const CONTENT_EXT = ".md";

export type ContentFile = {
  body: string;
  frontmatter: Record<string, unknown>;
};

function collectionDir(collection: Collection): string {
  return path.join(CONTENT_ROOT, collection);
}

export async function getContent(
  collection: Collection,
  slug: string
): Promise<ContentFile | null> {
  // The slug reaches this from a URL segment. Anything outside [a-z0-9-]
  // is rejected before it can become a path — no traversal, no absolute paths.
  if (!/^[a-z0-9-]+$/.test(slug)) return null;

  try {
    const raw = await readFile(
      path.join(collectionDir(collection), `${slug}${CONTENT_EXT}`),
      "utf8"
    );
    const { content, data } = matter(raw);
    return { body: content, frontmatter: data };
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === "ENOENT") return null;
    throw error;
  }
}

export function getProjectContent(slug: string): Promise<ContentFile | null> {
  return getContent("projects", slug);
}

export function getPostContent(slug: string): Promise<ContentFile | null> {
  return getContent("posts", slug);
}

export async function getContentSlugs(
  collection: Collection
): Promise<string[]> {
  try {
    const files = await readdir(collectionDir(collection));
    return files
      .filter((file) => file.endsWith(CONTENT_EXT))
      .map((file) => file.slice(0, -CONTENT_EXT.length));
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === "ENOENT") return [];
    throw error;
  }
}
```

```ts
import { inArray } from "drizzle-orm";
import { posts, postTags, tags } from "./schema";

// Postgres sorts NULLs FIRST on DESC, so an undated post would otherwise lead
// the blog. `nulls last` pushes it to the end; `id asc` is the stable tiebreak
// for posts sharing a date.
const postOrder = [
  sql`${posts.publishedAt} desc nulls last`,
  asc(posts.id),
];

const withPostTags = { postTags: { with: { tag: true } } } as const;

/** Post ids carrying a tag with this exact name. */
function postIdsWithTag(tagName: string) {
  return db
    .select({ id: postTags.postId })
    .from(postTags)
    .innerJoin(tags, eq(postTags.tagId, tags.id))
    .where(eq(tags.name, tagName));
}

export async function getPublishedPosts() {
  return db.query.posts.findMany({
    where: eq(posts.status, "published"),
    orderBy: postOrder,
    with: withPostTags,
  });
}

export type PublishedPost = Awaited<ReturnType<typeof getPublishedPosts>>[number];

export async function getPublishedPostsPage(
  page: number,
  perPage: number,
  tagName?: string
) {
  const where = tagName
    ? and(eq(posts.status, "published"), inArray(posts.id, postIdsWithTag(tagName)))
    : eq(posts.status, "published");

  const [rows, [totals]] = await Promise.all([
    db.query.posts.findMany({
      where,
      orderBy: postOrder,
      limit: perPage,
      offset: (page - 1) * perPage,
      with: withPostTags,
    }),
    db.select({ value: count() }).from(posts).where(where),
  ]);

  const total = totals?.value ?? 0;

  return {
    posts: rows,
    total,
    // At least 1, so an empty blog renders "page 1 of 1" rather than 404ing on
    // its own only valid page.
    totalPages: Math.max(1, Math.ceil(total / perPage)),
  };
}

export async function getFeaturedPosts(limit = 3) {
  return db.query.posts.findMany({
    where: and(eq(posts.status, "published"), eq(posts.featured, true)),
    orderBy: postOrder,
    limit,
    with: withPostTags,
  });
}

export async function getPostBySlug(slug: string) {
  return db.query.posts.findFirst({
    where: eq(posts.slug, slug),
    with: withPostTags,
  });
}

export async function getPublishedPostsCount(): Promise<number> {
  const [row] = await db
    .select({ value: count() })
    .from(posts)
    .where(eq(posts.status, "published"));
  return row?.value ?? 0;
}

/** Distinct tag names used by published posts, alphabetical. */
export async function getPostTagNames(): Promise<string[]> {
  const rows = await db
    .selectDistinct({ name: tags.name })
    .from(postTags)
    .innerJoin(tags, eq(postTags.tagId, tags.id))
    .innerJoin(posts, eq(postTags.postId, posts.id))
    .where(eq(posts.status, "published"))
    .orderBy(asc(tags.name));
  return rows.map((row) => row.name);
}
```

```ts
const WORDS_PER_MINUTE = 200;

export function readingMinutes(body: string): number {
  const words = body.trim().split(/\s+/).filter(Boolean).length;
  return Math.max(1, Math.ceil(words / WORDS_PER_MINUTE));
}

// `publishedAt` is a date-only column. Formatting it in the server's zone and
// again in the visitor's can land on different calendar days and trip a
// hydration mismatch, so the zone is pinned to UTC for everyone.
const dateFormat = new Intl.DateTimeFormat("en-US", {
  timeZone: "UTC",
  year: "numeric",
  month: "long",
  day: "numeric",
});

export function formatPostDate(
  value: string | Date | null | undefined
): string | null {
  if (!value) return null;
  const date = value instanceof Date ? value : new Date(`${value}T00:00:00Z`);
  if (Number.isNaN(date.getTime())) return null;
  return dateFormat.format(date);
}
```

```ts
type ImportedPost = {
  slug: string;
  title: string;
  excerpt: string;
  publishedAt: string | null;
  coverImage: string | null;
  featured: boolean;
  status: string;
  readingMinutes: number;

async function readPosts(): Promise<ImportedPost[]> {
  const slugs = await getContentSlugs("posts");
  const parsed: ImportedPost[] = [];

  for (const slug of slugs.sort()) {
    const content = await getPostContent(slug);
    if (!content) continue;

    const fm = content.frontmatter;
    const title = str(fm.title);
    const excerpt = str(fm.excerpt);

    if (!title || !excerpt) {
      console.warn(`  skipping ${slug}: missing ${!title ? "title" : "excerpt"}`);
      continue;
    }

    const postTagNames = Array.isArray(fm.tags)
      ? fm.tags.map((tag) => str(tag)).filter((tag): tag is string => tag !== null)
      : [];

    parsed.push({
      slug,
      title,
      excerpt,
      publishedAt: str(fm.publishedAt),
      coverImage: str(fm.coverImage),
      featured: fm.featured === true,
      status: str(fm.status) ?? "published",
      readingMinutes: readingMinutes(content.body),
      tags: postTagNames,
    });
  }

  return parsed;
}
```

```ts
export function tagSlug(name: string): string {
```

```tsx
import Link from "next/link";
import { notFound } from "next/navigation";

import { PostCard } from "@/components/blog/PostCard";

import { TagFilter } from "@/components/blog/TagFilter";
import {
  getFeaturedPosts,
  getPostTagNames,
  getPublishedPostsCount,
  getPublishedPostsPage,
} from "@/db/queries";

import { tagSlug } from "@/lib/tagSlug";
import styles from "./page.module.css";

export const POSTS_PER_PAGE = 6;

// The featured band's job is to surface picks out of a list too long to take in
// at once. Below one page's worth it just re-prints cards the reader can
// already see. Keep this equal to POSTS_PER_PAGE — the rule is "once the list
// fills a page", not "once there are six posts".
export const FEATURED_MIN_POSTS = 6;

type BlogPageProps = {
  searchParams: Promise<{ [key: string]: string | string[] | undefined }>;
};

/** `?page=` is untrusted input — anything that is not a positive integer is 1. */
function parsePage(value: string | string[] | undefined): number {
  const raw = Array.isArray(value) ? value[0] : value;
  if (!raw || !/^\d+$/.test(raw)) return 1;
  return Math.max(1, Number.parseInt(raw, 10));
}

function parseTag(value: string | string[] | undefined): string | undefined {
  const raw = Array.isArray(value) ? value[0] : value;
  return raw && raw.length > 0 ? raw : undefined;
}

export const metadata: Metadata = {
  title: "Blog — Sam Klepper",
  description: "Notes on building things with TypeScript, React and Postgres.",
};

export default async function BlogPage({ searchParams }: BlogPageProps) {
  const params = await searchParams;
  const page = parsePage(params.page);
  const requestedTag = parseTag(params.tag);

  const chips = (await getPostTagNames()).map((name) => ({
    name,
    slug: tagSlug(name),
  }));

  const activeTag = requestedTag
    ? chips.find((chip) => chip.slug === requestedTag)
    : undefined;

  if (requestedTag && !activeTag) notFound();

  const [{ posts, totalPages }, totalPublished] = await Promise.all([
    getPublishedPostsPage(page, POSTS_PER_PAGE, activeTag?.name),
    getPublishedPostsCount(),
  ]);

  if (page > totalPages) notFound();

  const bandAllowed =
    page === 1 && !activeTag && totalPublished >= FEATURED_MIN_POSTS;

  const featured = bandAllowed ? await getFeaturedPosts(3) : [];

  // Covers are resolved here rather than inside PostCard so the card stays a
  // synchronous component — an async child cannot be rendered by the tests.
  const featuredCards = await Promise.all(
    featured.map(async (post) => ({
      post,
      coverSrc: await resolvePublicImage(post.coverImage),
    }))
  );

  return (
    <main className={styles.page}>
      <Link href="/" className={styles.back}>
        <span aria-hidden="true">←</span> Home
      </Link>

      <h1 className={styles.heading}>Blog</h1>
      <p className={styles.intro}>
        Notes on building things — what broke, and what the fix actually was.
      </p>

      {featuredCards.length > 0 && (
        <section className={styles.featured} aria-labelledby="featured-heading">
          <h2 id="featured-heading" className={styles.featuredHeading}>
            Featured
          </h2>
          <div className={styles.featuredGrid}>
            {featuredCards.map(({ post, coverSrc }) => (
              <PostCard key={post.slug} post={post} coverSrc={coverSrc} />
            ))}
          </div>
        </section>
      )}

      <TagFilter tags={chips} activeSlug={activeTag?.slug} />

      <PostList posts={posts} />

      <Pagination
        page={page}
        totalPages={totalPages}
        basePath="/blog"
        label="Blog pagination"
      />
    </main>
  );
}
```

```tsx
export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
  const { slug } = await params;
  const post = await getPostBySlug(slug);

  if (!post || post.status !== "published") {
    return { title: "Not found — Sam Klepper" };
  }

  return {
    title: `${post.title} — Sam Klepper`,
    description: post.excerpt,
  };
}

export default async function BlogPostPage({ params }: PageProps) {
  const { slug } = await params;

  const [post, content] = await Promise.all([
    getPostBySlug(slug),
    getPostContent(slug),
  ]);

  if (!post || post.status !== "published" || !content) {
    notFound();
  }

  const tags = post.postTags.map((pt) => pt.tag);
  const date = formatPostDate(post.publishedAt);
  const cover = await resolvePublicImage(post.coverImage);

  return (
    <main className={styles.page}>
      <Link href="/blog" className={styles.back}>
        <span aria-hidden="true">←</span> Blog
      </Link>

      <header className={styles.header}>
        <p className={styles.eyebrow}>
          {date && <span>{date}</span>}
          {date && <span aria-hidden="true"> · </span>}
          <span>{post.readingMinutes} min read</span>
        </p>

        <h1 className={styles.title}>{post.title}</h1>
        <p className={styles.excerpt}>{post.excerpt}</p>

        {tags.length > 0 && (
          <ul className={styles.tags}>
            {tags.map((tag) => (
              <li key={tag.id} className={styles.tag}>
                {tag.name}
              </li>
            ))}
          </ul>
        )}
      </header>

      {cover && (
        <div className={styles.cover}>
          <Image
            src={cover}
            alt=""
            width={1536}
            height={864}
            className={styles.coverImage}
            sizes="(min-width: 768px) 48rem, 100vw"
            priority
          />
        </div>
      )}

      <article className={prose.prose}>
        <MDXRemote source={content.body} />
      </article>
    </main>
  );
}
```
